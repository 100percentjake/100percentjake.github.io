// ── MSAL + Graph ─────────────────────────────────────────────────────────────
function initMsal(){
  const redirectUri=window.location.href.split('?')[0].split('#')[0];
  const msalInstance=new msal.PublicClientApplication({
    auth:{clientId:CLIENT_ID,authority:'https://login.microsoftonline.com/organizations',redirectUri},
    cache:{cacheLocation:'localStorage'}
  });

  async function gAll(token,url,headers={}){
    let results=[],next=url;
    while(next){
      const r=await fetch(next,{headers:{Authorization:'Bearer '+token,...headers}});
      if(!r.ok)throw new Error(r.status);
      const p=await r.json();if(p.value)results=results.concat(p.value);next=p['@odata.nextLink']||null;
    }
    return results;
  }
  async function gCsv(token,url){const r=await fetch(url,{headers:{Authorization:'Bearer '+token}});if(!r.ok)return null;return r.text();}

  async function runReport(account){
    setLoading(true,'Loading...');
    document.getElementById('meta-user').textContent=account.username;
    document.getElementById('meta-tenant').textContent='Tenant: '+account.tenantId;
    document.getElementById('signout-btn').style.display='inline-flex';

    let token;
    try{token=(await msalInstance.acquireTokenSilent({scopes:SCOPES,account})).accessToken;}
    catch{msalInstance.loginRedirect({scopes:SCOPES,loginHint:account.username});return;}

    try{
      // ── 1. Users: try rich (signInActivity) first, fall back if 403 ────────
      setLoading(true,'Fetching users...');
      let rawUsers,signInAvailable=true;
      try{
        rawUsers=await gAll(token,
          'https://graph.microsoft.com/v1.0/users?$select=id,displayName,userPrincipalName,mail,accountEnabled,assignedLicenses,signInActivity,userType,jobTitle,department&$top=999',
          {'ConsistencyLevel':'eventual'});
      }catch(e){
        if(String(e.message).includes('403')){
          signInAvailable=false;
          recordDegradation(
            'signInActivity',
            'Last Sign-in dates unavailable',
            'This tenant does not have Entra ID P1 or P2 licensing, which is required to read signInActivity. Sign-in dates and stale account detection are disabled.'
          );
          rawUsers=await gAll(token,
            'https://graph.microsoft.com/v1.0/users?$select=id,displayName,userPrincipalName,mail,accountEnabled,assignedLicenses,userType,jobTitle,department&$top=999');
        }else{throw e;}
      }

      // ── 2. Groups + Teams ───────────────────────────────────────────────────
      setLoading(true,'Fetching groups...');
      const [groups,teamsRaw]=await Promise.all([
        gAll(token,'https://graph.microsoft.com/v1.0/groups?$select=id,displayName,mail,mailEnabled,securityEnabled,groupTypes,description&$top=999'),
        gAll(token,"https://graph.microsoft.com/v1.0/groups?$filter=resourceProvisioningOptions/Any(x:x eq 'Team')&$select=id&$top=999")
      ]);
      const teamsSet=new Set(teamsRaw.map(t=>t.id));

      // ── 3. Directory roles: fail-safe ────────────────────────────────────
      setLoading(true,'Fetching admin roles...');
      const userRoleMap={};
      try{
        const roles=await gAll(token,'https://graph.microsoft.com/v1.0/directoryRoles?$select=id,displayName');
        await Promise.all(roles.map(async r=>{
          try{
            const members=await gAll(token,`https://graph.microsoft.com/v1.0/directoryRoles/${r.id}/members?$select=id`);
            members.forEach(m=>{if(!userRoleMap[m.id])userRoleMap[m.id]=[];userRoleMap[m.id].push(r.displayName);});
          }catch{}
        }));
      }catch(e){
        if(String(e.message).includes('403')){
          recordDegradation(
            'directoryRoles',
            'Admin role assignments unavailable',
            'Directory role enumeration was blocked (403). Admin consent for Directory.Read.All may not have been granted. Role pills will not appear on user entries.'
          );
        }
      }

      // ── 4. Group activity report: fail-safe ─────────────────────────────
      setLoading(true,'Fetching activity data...');
      const activityMap={};
      try{
        const csv=await gCsv(token,"https://graph.microsoft.com/v1.0/reports/getOffice365GroupsActivityDetail(period='D180')");
        if(csv){
          const lines=csv.split('\n'),hdr=lines[0].split(',');
          const ii=hdr.findIndex(h=>h.replace(/"/g,'').trim()==='Group Id');
          const li=hdr.findIndex(h=>h.replace(/"/g,'').trim()==='Last Activity Date');
          for(let i=1;i<lines.length;i++){const c=lines[i].split(',');const gid=(c[ii]||'').replace(/"/g,'').trim();const lad=(c[li]||'').replace(/"/g,'').trim();if(gid)activityMap[gid]=lad||null;}
        }
      }catch(e){
        if(String(e.message).includes('403')){
          recordDegradation(
            'groupActivity',
            'Group activity dates unavailable',
            'The Groups activity report could not be retrieved (403). Last activity chips on resource cards will show "No activity data". Ensure the signed-in account has the Reports Reader role.'
          );
        }
      }

      // ── 5. Per-group members + owners ───────────────────────────────────
      const resources=[];
      for(let i=0;i<groups.length;i++){
        const g=groups[i];setLoading(true,`Fetching members (${i+1}/${groups.length})...`);
        let members=[],owners=[];
        try{members=await gAll(token,`https://graph.microsoft.com/v1.0/groups/${g.id}/members/microsoft.graph.user?$select=id,displayName,userPrincipalName,mail,accountEnabled,assignedLicenses&$top=999`);}catch{}
        try{owners=await gAll(token,`https://graph.microsoft.com/v1.0/groups/${g.id}/owners/microsoft.graph.user?$select=id,displayName,userPrincipalName,mail,accountEnabled,assignedLicenses&$top=999`);}catch{}
        const type=teamsSet.has(g.id)?'Teams-enabled Group':(g.groupTypes||[]).includes('Unified')?'M365 Group':(!!g.mailEnabled&&!!g.securityEnabled)?'Mail-enabled Security':g.mailEnabled?'Distribution List':g.securityEnabled?'Security Group':'Group';
        resources.push({id:g.id,displayName:g.displayName,mail:g.mail,description:g.description,type,lastActivity:activityMap[g.id]??null,members,owners});
      }

      // ── 6. Bake users, infer shared mailboxes ───────────────────────────
      rawUsers.forEach(u=>{
        u._roles=userRoleMap[u.id]||[];
        if(inferSharedMailbox(u)){
          u._isSharedMailbox=true;
          resources.push({id:u.id,displayName:u.displayName,mail:u.mail||u.userPrincipalName,description:'Shared Mailbox (inferred)',type:'Shared Mailbox',lastActivity:null,members:[],owners:[],delegatesNote:true});
        }
      });

      allResources=resources;
      allUsers=rawUsers.map(bakeUserLicenses);

      // ── 7. Render + show capability notices if any ──────────────────────
      renderCapabilityBar();
      setLoading(false);show('report-screen');renderResources();renderUsers();

    }catch(e){setLoading(false);show('signin-screen');showError('Report failed: '+(e.message||e));}
  }

  msalInstance.handleRedirectPromise()
    .then(r=>{
      if(r&&r.account)runReport(r.account);
      else{const a=msalInstance.getAllAccounts();if(a.length)runReport(a[0]);else{setLoading(false);show('signin-screen');}}
    })
    .catch(e=>{setLoading(false);show('signin-screen');showError(e.message||'Authentication error');});

  document.getElementById('signin-btn').addEventListener('click',()=>msalInstance.loginRedirect({scopes:SCOPES,prompt:'select_account'}));
  document.getElementById('signout-btn').addEventListener('click',()=>msalInstance.logoutRedirect({postLogoutRedirectUri:redirectUri}));
}
</script>
